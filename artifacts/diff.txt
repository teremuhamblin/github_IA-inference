+def format_diff(diff_content: str) -> str:
+    """
+    Formate le contenu d’un diff pour l’analyse IA.
+    """
+    lines = diff_content.split("\n")
+    cleaned = [line for line in lines if not line.startswith("index")]
+    return "\n".join(cleaned)

+def extract_added_lines(diff_content: str):
+    return [line for line in diff_content.split("\n") if line.startswith("+")]

+def extract_removed_lines(diff_content: str):
+    return [line for line in diff_content.split("\n") if line.startswith("-")]
